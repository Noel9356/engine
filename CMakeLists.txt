option(
    ENGINE_ENABLE_CCACHE
    "Use ccache for C and C++ compilation when available"
    ON
)
if(ENGINE_ENABLE_CCACHE)
    find_program(
        ENGINE_CCACHE_EXECUTABLE
        NAMES ccache
    )
    if(ENGINE_CCACHE_EXECUTABLE)
        if(NOT
           CMAKE_C_COMPILER_LAUNCHER
        )
            set(CMAKE_C_COMPILER_LAUNCHER
                "${ENGINE_CCACHE_EXECUTABLE}"
                CACHE STRING
                      "C compiler launcher"
                      FORCE
            )
        endif()
        if(NOT
           CMAKE_CXX_COMPILER_LAUNCHER
        )
            set(CMAKE_CXX_COMPILER_LAUNCHER
                "${CMAKE_COMMAND};-E;env;CCACHE_SLOPPINESS=pch_defines,time_macros;${ENGINE_CCACHE_EXECUTABLE}"
                CACHE STRING
                      "C++ compiler launcher"
                      FORCE
            )
            if(CMAKE_CXX_COMPILER_ID
               STREQUAL
               "GNU"
            )
                add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-fpch-preprocess>")
            endif()
        endif()
        message(STATUS "engine: compiler cache enabled (${ENGINE_CCACHE_EXECUTABLE})")
    else()
        message(STATUS "engine: ccache not found; compiler caching is disabled")
    endif()
endif()

set(ENGINE_GCC_PGO_MODE
    "OFF"
    CACHE STRING
          "GCC profile-guided optimization mode: OFF, GENERATE, or USE"
)

    CACHE ENGINE_GCC_PGO_MODE
    PROPERTY STRINGS
             OFF
             GENERATE
             USE
)
set(ENGINE_GCC_PGO_DIRECTORY
    "${CMAKE_BINARY_DIR}/pgo_data"
    CACHE PATH
          "Directory containing GCC profile data generated by ENGINE_GCC_PGO_MODE=GENERATE"
)

string(
    TOUPPER "${ENGINE_GCC_PGO_MODE}"
            engineGccPgoMode
)
if(NOT
   engineGccPgoMode
   MATCHES
   "^(OFF|GENERATE|USE)$"
)
    message(FATAL_ERROR "ENGINE_GCC_PGO_MODE must be OFF, GENERATE, or USE (got '${ENGINE_GCC_PGO_MODE}')")
endif()

if(ENGINE_NATIVE_CPU_OPTIMIZATION)
    if(CMAKE_CXX_COMPILER_ID
       MATCHES
       "GNU|Clang|AppleClang"
    )
        add_compile_options("$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:C,CXX>>:-march=native>")
        message(STATUS "engine: native CPU tuning enabled for Release; binaries are machine-specific")
    else()
        message(FATAL_ERROR "ENGINE_NATIVE_CPU_OPTIMIZATION is currently supported only by GCC and Clang toolchains")
    endif()
endif()

if(NOT
   engineGccPgoMode
   STREQUAL
   "OFF"
)
    if(NOT
       CMAKE_CXX_COMPILER_ID
       STREQUAL
       "GNU"
       OR NOT
          CMAKE_C_COMPILER_ID
          STREQUAL
          "GNU"
    )
        message(FATAL_ERROR "engine: ENGINE_GCC_PGO_MODE requires GCC for both C and C++")
    endif()
    get_filename_component(
        engineGccPgoDirectory
        "${ENGINE_GCC_PGO_DIRECTORY}"
        ABSOLUTE
    )
    if(engineGccPgoMode
       STREQUAL
       "GENERATE"
    )
        file(
            MAKE_DIRECTORY
            "${engineGccPgoDirectory}"
        )
        add_compile_options(
            "$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:C,CXX>>:-fprofile-generate=${engineGccPgoDirectory}>"
        )
        add_link_options("$<$<CONFIG:Release>:-fprofile-generate=${engineGccPgoDirectory}>")
        message(STATUS "engine: GCC PGO instrumentation enabled; profile data directory: ${engineGccPgoDirectory}")
    else()
        if(NOT
           IS_DIRECTORY
           "${engineGccPgoDirectory}"
        )
            message(
                FATAL_ERROR
                    "engine: GCC PGO profile directory does not exist: ${engineGccPgoDirectory}. Run a GENERATE build first."
            )
        endif()
        file(
            GLOB_RECURSE
            engineGccPgoProfiles
            LIST_DIRECTORIES false
            "${engineGccPgoDirectory}/*.gcda"
        )
        if(NOT
           engineGccPgoProfiles
        )
            message(
                FATAL_ERROR
                    "engine: no GCC .gcda profile data found in ${engineGccPgoDirectory}. Run the instrumented application before configuring PGO USE."
            )
        endif()
        add_compile_options(
            "$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:C,CXX>>:-fprofile-use=${engineGccPgoDirectory}>"
            "$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:C,CXX>>:-fprofile-correction>"
        )
        message(STATUS "engine: GCC profile-guided optimization enabled from ${engineGccPgoDirectory}")
    endif()
    if(CMAKE_C_COMPILER_LAUNCHER
       OR CMAKE_CXX_COMPILER_LAUNCHER
    )
        message(
            WARNING
                "engine: GCC PGO is more reliable with compiler caching disabled; configure with -DENGINE_ENABLE_CCACHE=OFF and clear any externally configured compiler launcher."
        )
    endif()
endif()

if(ENGINE_ENABLE_SANITIZERS)
    if(MSVC)
        add_compile_options(/fsanitize=address)
        add_link_options(/INCREMENTAL:NO)
        message(STATUS "engine: AddressSanitizer enabled (MSVC; UBSan is not available)")
    elseif(
        CMAKE_CXX_COMPILER_ID
        MATCHES
        "GNU|Clang|AppleClang"
    )
        add_compile_options(
            -fsanitize=address,undefined
            -fno-omit-frame-pointer
        )
        add_link_options(-fsanitize=address,undefined)
        message(STATUS "engine: AddressSanitizer and UndefinedBehaviorSanitizer enabled")
    else()
        message(FATAL_ERROR "ENGINE_ENABLE_SANITIZERS=ON is unsupported by compiler '${CMAKE_CXX_COMPILER_ID}'")
    endif()
endif()

if(ENGINE_ENABLE_LTO)
    if(WIN32)
        message(
            STATUS
                "engine: ENGINE_ENABLE_LTO=ON is not applied on Windows (the kernel's symbol-export mechanism does not compose with whole-program optimization)"
        )
    elseif(
        CMAKE_BUILD_TYPE
        MATCHES
        "Release|MinSizeRel|RelWithDebInfo"
    )
        include(CheckIPOSupported)
        check_ipo_supported(
            RESULT engineIpoSupported
            OUTPUT engineIpoError
        )
        if(engineIpoSupported)
            set(CMAKE_INTERPROCEDURAL_OPTIMIZATION
                ON
            )
            message(STATUS "engine: link-time optimization enabled for this build")
        else()
            message(
                FATAL_ERROR "engine: ENGINE_ENABLE_LTO=ON but this toolchain does not support IPO (${engineIpoError})"
            )
        endif()
    endif()
endif()

foreach(
    requestedModule
    IN
    LISTS ENGINE_SELECTED_MODULES
)
    set(normalizedModule
        "${requestedModule}"
    )
    foreach(
        aliasPair
        IN
        LISTS ENGINE_MODULE_SELECTION_ALIASES
    )
        string(
            REPLACE "="
                    ";"
                    aliasParts
                    "${aliasPair}"
        )
        list(
            GET
            aliasParts
            0
            aliasName
        )
        list(
            GET
            aliasParts
            1
            canonicalName
        )
        if(requestedModule
           STREQUAL
           aliasName
        )
            set(normalizedModule
                "${canonicalName}"
            )
            break()
        endif()
    endforeach()
    list(
        APPEND
        ENGINE_NORMALIZED_MODULES
        "${normalizedModule}"
    )
endforeach()

set(engineSelectionChanged
    TRUE
)
while(engineSelectionChanged)
    set(engineSelectionChanged
        FALSE
    )
    foreach(
        selectedModule
        IN
        LISTS ENGINE_SELECTED_MODULES
    )
        set(moduleDirectoryVariable
            "ENGINE_MODULE_DIR_${selectedModule}"
        )
        set(moduleDirectory
            "${${moduleDirectoryVariable}}"
        )
        set(moduleCmake
            "${moduleDirectory}/CMakeLists.txt"
        )
        if(NOT
           EXISTS
           "${moduleCmake}"
        )
            continue()
        endif()
        file(
            READ
            "${moduleCmake}"
            moduleCmakeContent
        )
        string(
            REGEX MATCHALL
                  "engine_[A-Za-z0-9_]+_capabilities"
                  capabilityTargets
                  "${moduleCmakeContent}"
        )
        foreach(
            capabilityTarget
            IN
            LISTS capabilityTargets
        )
            string(
                REGEX
                REPLACE "^engine_(.+)_capabilities$"
                        "\\1"
                        dependencyModule
                        "${capabilityTarget}"
            )
            foreach(
                aliasPair
                IN
                LISTS ENGINE_MODULE_SELECTION_ALIASES
            )
                string(
                    REPLACE "="
                            ";"
                            aliasParts
                            "${aliasPair}"
                )
                list(
                    GET
                    aliasParts
                    0
                    aliasName
                )
                list(
                    GET
                    aliasParts
                    1
                    canonicalName
                )
                if(dependencyModule
                   STREQUAL
                   aliasName
                )
                    set(dependencyModule
                        "${canonicalName}"
                    )
                    break()
                endif()
            endforeach()
            if(dependencyModule
               IN_LIST
               ENGINE_AVAILABLE_MODULES
               AND NOT
                   dependencyModule
                   IN_LIST
                   ENGINE_SELECTED_MODULES
            )
                list(
                    APPEND
                    ENGINE_SELECTED_MODULES
                    "${dependencyModule}"
                )
                set(engineSelectionChanged
                    TRUE
                )
            endif()
        endforeach()
    endforeach()
endwhile()

        )
    endif()
endforeach()

    gameModPackageDirectory
    IN
    LISTS ENGINE_GAME_MOD_PACKAGE_DIRS
)
    if(NOT
       EXISTS
       "${gameModPackageDirectory}/mod.toml"
    )
        continue()
    endif()
    get_filename_component(
        gameModPackageId
        "${gameModPackageDirectory}"
        NAME
    )
    if(EXISTS
       "${gameModPackageDirectory}/CMakeLists.txt"
    )
        set(gameModBuildTarget
            "engine_game_mod_${gameModPackageId}"
        )
        if(NOT
           TARGET
           "${gameModBuildTarget}"
        )
            message(
                STATUS
                    "engine: skipping native mod package '${gameModPackageId}' because its build target is unavailable"
            )
            continue()
        endif()
    endif()
    list(
        APPEND
        ENGINE_GAME_MOD_PACKAGE_COMMANDS
        COMMAND
        ${CMAKE_COMMAND}
        -E
        make_directory
        "${CMAKE_BINARY_DIR}/runtime/mods/${gameModPackageId}"
        COMMAND
        ${CMAKE_COMMAND}
        -E
        copy_if_different
        "${gameModPackageDirectory}/mod.toml"
        "${CMAKE_BINARY_DIR}/runtime/mods/${gameModPackageId}/mod.toml"
    )
    foreach(
        packageFile
        config.toml
        scene.toml
    )
        if(EXISTS
           "${gameModPackageDirectory}/${packageFile}"
        )
            list(
                APPEND
                ENGINE_GAME_MOD_PACKAGE_COMMANDS
                COMMAND
                ${CMAKE_COMMAND}
                -E
                copy_if_different
                "${gameModPackageDirectory}/${packageFile}"
                "${CMAKE_BINARY_DIR}/runtime/mods/${gameModPackageId}/${packageFile}"
            )
        endif()
    endforeach()

        GLOB
        gameModPackageScripts

    )

        gameModScript
        IN

            APPEND
            ENGINE_GAME_MOD_PACKAGE_COMMANDS
            COMMAND
            ${CMAKE_COMMAND}

            copy_if_different

            "${CMAKE_BINARY_DIR}/runtime/mods/${gameModPackageId}/"

       "${gameModPackageDirectory}/assets"
    )

            APPEND
            ENGINE_GAME_MOD_PACKAGE_COMMANDS
            COMMAND
            ${CMAKE_COMMAND}
            -E
            copy_directory
            "${gameModPackageDirectory}/assets"
            "${CMAKE_BINARY_DIR}/runtime/mods/${gameModPackageId}/assets"
        )
